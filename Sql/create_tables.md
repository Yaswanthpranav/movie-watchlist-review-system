
## set of queries ran to create the tables

-- 1. LANGUAGE
CREATE TABLE public.language (
    language_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    language_name VARCHAR(50) NOT NULL UNIQUE
);

-- 2. GENRE
CREATE TABLE public.genre (
    genre_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    genre_name VARCHAR(50) NOT NULL UNIQUE
);

-- 3. STREAMING_PLATFORM
CREATE TABLE public.streaming_platform (
    streaming_platform_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    streaming_platform_name VARCHAR(100) NOT NULL UNIQUE
);

-- 4. USER
-- Quoted because USER is a PostgreSQL keyword.
CREATE TABLE public."user" (
    user_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    create_date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- 5. MOVIE
CREATE TABLE public.movie (
    movie_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    release_yr SMALLINT,
    duration INTEGER NOT NULL,
    language_id INTEGER NOT NULL,
    description TEXT,
    age_rating VARCHAR(10),

    CONSTRAINT movie_language_fk
        FOREIGN KEY (language_id)
        REFERENCES public.language(language_id),

    CONSTRAINT movie_duration_check
        CHECK (duration > 0)
);

-- 6. MOVIE_GENRE
CREATE TABLE public.movie_genre (
    movie_id INTEGER NOT NULL,
    genre_id INTEGER NOT NULL,

    PRIMARY KEY (movie_id, genre_id),

    CONSTRAINT movie_genre_movie_fk
        FOREIGN KEY (movie_id)
        REFERENCES public.movie(movie_id)
        ON DELETE CASCADE,

    CONSTRAINT movie_genre_genre_fk
        FOREIGN KEY (genre_id)
        REFERENCES public.genre(genre_id)
        ON DELETE CASCADE
);

-- 7. MOVIE_PLATFORM
CREATE TABLE public.movie_platform (
    movie_id INTEGER NOT NULL,
    platform_id INTEGER NOT NULL,

    PRIMARY KEY (movie_id, platform_id),

    CONSTRAINT movie_platform_movie_fk
        FOREIGN KEY (movie_id)
        REFERENCES public.movie(movie_id)
        ON DELETE CASCADE,

    CONSTRAINT movie_platform_platform_fk
        FOREIGN KEY (platform_id)
        REFERENCES public.streaming_platform(streaming_platform_id)
        ON DELETE CASCADE
);

-- 8. WATCHLIST
CREATE TABLE public.watchlist (
    watchlist_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id INTEGER NOT NULL,
    watchlist_name VARCHAR(100) NOT NULL,
    notes TEXT,
    create_date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT watchlist_user_name_unique
        UNIQUE (user_id, watchlist_name),

    CONSTRAINT watchlist_user_fk
        FOREIGN KEY (user_id)
        REFERENCES public."user"(user_id)
        ON DELETE CASCADE
);

-- 9. WATCHLIST_ITEMS
CREATE TABLE public.watchlist_items (
    watchlist_item_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    watchlist_id INTEGER NOT NULL,
    movie_id INTEGER NOT NULL,
    added_date TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'PLANNED',

    CONSTRAINT watchlist_movie_unique
        UNIQUE (watchlist_id, movie_id),

    CONSTRAINT watchlist_items_watchlist_fk
        FOREIGN KEY (watchlist_id)
        REFERENCES public.watchlist(watchlist_id)
        ON DELETE CASCADE,

    CONSTRAINT watchlist_items_movie_fk
        FOREIGN KEY (movie_id)
        REFERENCES public.movie(movie_id),

    CONSTRAINT watchlist_status_check
        CHECK (status IN ('PLANNED', 'WATCHING', 'COMPLETED'))
);

-- 10. REVIEWS
CREATE TABLE public.reviews (
    review_id INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id INTEGER NOT NULL,
    movie_id INTEGER NOT NULL,
    platform_id INTEGER,
    rating DECIMAL(2,1) NOT NULL,
    review_text TEXT,
    reviewed_on TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT reviews_user_movie_unique
        UNIQUE (user_id, movie_id),

    CONSTRAINT reviews_user_fk
        FOREIGN KEY (user_id)
        REFERENCES public."user"(user_id),

    CONSTRAINT reviews_movie_fk
        FOREIGN KEY (movie_id)
        REFERENCES public.movie(movie_id),

    CONSTRAINT reviews_platform_fk
        FOREIGN KEY (platform_id)
        REFERENCES public.streaming_platform(streaming_platform_id),

    CONSTRAINT reviews_rating_check
        CHECK (rating BETWEEN 1 AND 5)
);

CREATE INDEX idx_movie_language
ON public.movie(language_id);

CREATE INDEX idx_movie_genre_genre
ON public.movie_genre(genre_id);

CREATE INDEX idx_movie_platform_platform
ON public.movie_platform(platform_id);

CREATE INDEX idx_watchlist_items_movie
ON public.watchlist_items(movie_id);

CREATE INDEX idx_reviews_movie
ON public.reviews(movie_id);

CREATE INDEX idx_reviews_platform
ON public.reviews(platform_id);
